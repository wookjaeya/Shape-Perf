# the script (<scratch>/8cfix/pack.sh)
  #!/bin/bash
  # pack.sh SCRATCH ITEM...: for each item (a directory or file under SCRATCH), write a sha256 list of its files,
  # a tar.xz of it, check by streaming the archive that every file has the same sha256, and only then remove the item.
  set -u
  S=$1; shift
  cd "$S" || exit 2
  for it in "$@"; do
    [ -e "$it" ] || { echo "$it: missing"; continue; }
    b=$(du -s -B1 "$it" | cut -f1)
    find "$it" -type f -print0 | sort -z | xargs -0 sha256sum > "8cfix/pack/$it.sha256" || { echo "$it: hash failed"; exit 1; }
    tar -cf - "$it" | xz -6 -T2 > "8cfix/pack/$it.tar.xz" || { echo "$it: pack failed"; exit 1; }
    xz -dc "8cfix/pack/$it.tar.xz" | tar -xf - --to-command='sha256sum | sed "s|-\$|$TAR_FILENAME|"' | sort -k2 > "8cfix/pack/$it.check"
    if cmp -s <(sort -k2 "8cfix/pack/$it.sha256") "8cfix/pack/$it.check"; then
      n=$(wc -l < "8cfix/pack/$it.sha256"); z=$(stat -c %s "8cfix/pack/$it.tar.xz")
      rm -rf "$it" && echo "$it: $n files, $b B -> $z B xz, stream check same, removed"
    else
      echo "$it: CHECK DIFFERS, kept"; exit 1
    fi
  done
