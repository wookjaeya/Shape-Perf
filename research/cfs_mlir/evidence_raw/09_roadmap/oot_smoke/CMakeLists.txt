cmake_minimum_required(VERSION 3.20)
project(cfs_smoke LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 17)
find_package(MLIR REQUIRED CONFIG)
list(APPEND CMAKE_MODULE_PATH "${MLIR_CMAKE_DIR}" "${LLVM_CMAKE_DIR}")
include(TableGen)
include(AddLLVM)
include(AddMLIR)
include_directories(${LLVM_INCLUDE_DIRS} ${MLIR_INCLUDE_DIRS} ${CMAKE_CURRENT_BINARY_DIR}/include)
set(LLVM_TARGET_DEFINITIONS include/CfsOps.td)
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/include)
mlir_tablegen(include/CfsOps.h.inc -gen-op-decls)
mlir_tablegen(include/CfsOps.cpp.inc -gen-op-defs)
mlir_tablegen(include/CfsDialect.h.inc -gen-dialect-decls -dialect=cfs)
mlir_tablegen(include/CfsDialect.cpp.inc -gen-dialect-defs -dialect=cfs)
add_public_tablegen_target(CfsIncGen)
add_executable(cfs-smoke tools/cfs-smoke.cpp lib/Cfs.cpp)
add_dependencies(cfs-smoke CfsIncGen)
target_link_libraries(cfs-smoke PRIVATE MLIRIR MLIRParser MLIRArithDialect MLIRSideEffectInterfaces MLIRSupport)
add_executable(dense-smoke tools/dense-smoke.cpp)
target_link_libraries(dense-smoke PRIVATE MLIRAnalysis MLIRLLVMDialect MLIRDLTIDialect MLIRParser MLIRIR MLIRSupport)
